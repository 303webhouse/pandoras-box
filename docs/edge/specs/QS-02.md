QS-02 — Phase 0: outcome coverage done right + timestamp-offset profile
Context: QS-01 established signal_id (varchar) exists on BOTH tables as the
natural key; outcomes rows appear to be placeholder-inserted at signal
creation; signals carries a parallel outcome system. Census, not a
hypothesis test — no pre-registration required.

-- QS-02-A1: key format sample, signals side
SELECT signal_id FROM signals ORDER BY id DESC LIMIT 5;

-- QS-02-A2: key format sample, outcomes side
SELECT signal_id FROM signal_outcomes ORDER BY id DESC LIMIT 5;

-- QS-02-A3: key uniqueness, signals
SELECT COUNT(*) AS n, COUNT(DISTINCT signal_id) AS n_distinct FROM signals;

-- QS-02-A4: key uniqueness, outcomes
SELECT COUNT(*) AS n, COUNT(DISTINCT signal_id) AS n_distinct
FROM signal_outcomes;

-- QS-02-A5: match rate on the natural key
SELECT COUNT(*) AS n_outcome_rows, COUNT(s.signal_id) AS matched_to_signals
FROM signal_outcomes o LEFT JOIN signals s ON s.signal_id = o.signal_id;

-- QS-02-B1: outcomes.outcome value distribution
SELECT outcome, COUNT(*) AS n FROM signal_outcomes
GROUP BY 1 ORDER BY 2 DESC LIMIT 30;

-- QS-02-B2: resolution coverage per strategy x direction (fan-out safe)
SELECT s.strategy, s.direction,
       COUNT(DISTINCT s.signal_id) AS n_signals,
       COUNT(DISTINCT o.signal_id) AS n_outcome_rows,
       COUNT(DISTINCT CASE WHEN o.outcome IS NOT NULL
             THEN o.signal_id END) AS n_resolved,
       ROUND(100.0 * COUNT(DISTINCT CASE WHEN o.outcome IS NOT NULL
             THEN o.signal_id END)
             / NULLIF(COUNT(DISTINCT s.signal_id),0), 1) AS pct_resolved
FROM signals s LEFT JOIN signal_outcomes o ON o.signal_id = s.signal_id
GROUP BY 1, 2 ORDER BY 3 DESC LIMIT 60;

-- QS-02-B3: signals-side (parallel system) coverage
SELECT strategy, direction, COUNT(*) AS n,
       COUNT(outcome) AS n_sig_outcome,
       COUNT(outcome_resolved_at) AS n_resolved_at
FROM signals GROUP BY 1, 2 ORDER BY 3 DESC LIMIT 60;

-- QS-02-B4: signals.outcome value distribution
SELECT outcome, COUNT(*) AS n FROM signals
GROUP BY 1 ORDER BY 2 DESC LIMIT 30;

-- QS-02-B5: divergence log size
SELECT COUNT(*) AS n_rows FROM signal_outcome_diff_log;

-- QS-02-B6: resolver liveness
SELECT COUNT(outcome_at) AS n_with_outcome_at,
       MIN(outcome_at) AS min_oa, MAX(outcome_at) AS max_oa
FROM signal_outcomes;

-- QS-02-C1: placeholder-insert test (creation lag)
SELECT COUNT(*) AS n_pairs,
       COUNT(*) FILTER (WHERE o.created_at - s.created_at
                        < INTERVAL '5 seconds') AS created_within_5s
FROM signal_outcomes o JOIN signals s ON s.signal_id = o.signal_id;

-- QS-02-D1: weekly modal offset, stamped hour vs hour_of_day column
SELECT DATE_TRUNC('week', created_at) AS wk,
       MODE() WITHIN GROUP (ORDER BY
         (EXTRACT(HOUR FROM created_at)::int - hour_of_day + 24) % 24)
         AS modal_offset,
       COUNT(*) AS n
FROM signals WHERE hour_of_day IS NOT NULL
GROUP BY 1 ORDER BY 1 LIMIT 40;

-- QS-02-D2: weekly created_at minus "timestamp" delta
SELECT DATE_TRUNC('week', created_at) AS wk,
       ROUND((AVG(EXTRACT(EPOCH FROM (created_at - "timestamp"))/3600))
             ::numeric, 2) AS avg_hours_diff,
       COUNT(*) AS n
FROM signals GROUP BY 1 ORDER BY 1 LIMIT 40;

-- QS-02-D3: stamped-hour histogram, weekday-only strategies, by month
SELECT DATE_TRUNC('month', created_at) AS mo,
       EXTRACT(HOUR FROM created_at)::int AS hr, COUNT(*) AS n
FROM signals
WHERE strategy IN ('Holy_Grail','CTA Scanner','sell_the_rip',
                   'Footprint_Imbalance','Artemis')
GROUP BY 1, 2 ORDER BY 1, 2 LIMIT 200;

-- QS-02-E0: column ground truth, round 2
SELECT table_name, ordinal_position, column_name, data_type
FROM information_schema.columns
WHERE table_schema = 'public'
  AND table_name IN ('signal_outcome_diff_log','signal_forward_returns',
      'strategy_health','triton_flow_shadow','background_task_failures',
      'price_history','committee_passes','rh_trade_history')
ORDER BY table_name, ordinal_position LIMIT 400;

-- QS-02-E1
SELECT COUNT(*) AS n FROM background_task_failures;
-- QS-02-E2
SELECT COUNT(*) AS n FROM price_history;
-- QS-02-E3
SELECT COUNT(*) AS n FROM strategy_health;
-- QS-02-E4
SELECT COUNT(*) AS n FROM triton_flow_shadow;

-- QS-02-F1: strategy x signal_type map (Rosetta raw material)
SELECT strategy, signal_type, COUNT(*) AS n
FROM signals GROUP BY 1, 2 ORDER BY 1, 3 DESC LIMIT 100;

-- QS-02-F2: strategy x source map (delivery-path evidence)
SELECT strategy, source, COUNT(*) AS n
FROM signals GROUP BY 1, 2 ORDER BY 1, 3 DESC LIMIT 100;

END QS-02
