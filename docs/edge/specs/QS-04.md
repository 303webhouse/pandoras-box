-- QS-04-1: drill to_char exhibit (lens-immune text render)
SELECT id, to_char(created_at,'YYYY-MM-DD HH24:MI:SS') AS stored_text
FROM bias_composite_history WHERE id IN (27774, 27777);

-- QS-04-2 (G1v2): resolution mix, status-stratified
SELECT s.strategy, s.direction, s.status, o.outcome, COUNT(*) AS n
FROM signals s
LEFT JOIN signal_outcomes o ON o.signal_id = s.signal_id
GROUP BY 1,2,3,4 ORDER BY 1,2,3,4 LIMIT 500;

-- QS-04-3 (ORPH-3): stem-twin recovery attribution
SELECT s2.strategy, o.signal_type, COUNT(*) AS n
FROM signal_outcomes o
LEFT JOIN signals s ON s.signal_id = o.signal_id
JOIN signals s2
  ON regexp_replace(s2.signal_id,'_[^_]+$','') =
     regexp_replace(o.signal_id,'_[^_]+$','')
WHERE s.signal_id IS NULL
GROUP BY 1,2 ORDER BY 3 DESC LIMIT 40;

-- QS-04-4: uncovered signals (no outcome row), era-dated
SELECT s.strategy, DATE_TRUNC('month', s.created_at) AS mo, COUNT(*) AS n
FROM signals s
LEFT JOIN signal_outcomes o ON o.signal_id = s.signal_id
WHERE o.signal_id IS NULL
GROUP BY 1,2 ORDER BY 1,2 LIMIT 100;

-- QS-04-5: crypto_engine liveness dating (FRF/Liq question)
SELECT s.created_at::date AS d, s.strategy, s.signal_type
FROM signals s WHERE s.source = 'crypto_engine'
ORDER BY 1 LIMIT 20;

-- QS-04-6a/6b: health_alerts census
SELECT COUNT(*) AS n FROM health_alerts;
SELECT column_name, data_type FROM information_schema.columns
WHERE table_schema='public' AND table_name='health_alerts'
ORDER BY ordinal_position LIMIT 60;

-- QS-04-7a/7b/7c: trades-store recon (Phase-1 realized side)
SELECT COUNT(*) AS n_trades, COUNT(DISTINCT signal_id) AS n_signals
FROM trades;
SELECT COUNT(*) AS n_legs FROM trade_legs;
SELECT column_name, data_type FROM information_schema.columns
WHERE table_schema='public' AND table_name IN ('trades','trade_legs')
ORDER BY table_name, ordinal_position LIMIT 120;

END QS-04
