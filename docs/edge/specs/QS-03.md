QS-03 — Phase 0 closeout: per-table clock verdicts, diff-log forensics,
shadow/health extents, Phase-1 raw crosstab. Census, not hypothesis test.

-- QS-03-A0: name the lens
SELECT current_setting('TimeZone') AS session_tz;

-- QS-03-A1: signals write-clock adjudicator
SELECT (NOW() AT TIME ZONE 'UTC') AS utc_now_naive,
       MAX(created_at) AS max_created,
       (NOW() AT TIME ZONE 'UTC') - MAX(created_at) AS age
FROM signals;

-- QS-03-A2: signal_outcomes adjudicator
SELECT (NOW() AT TIME ZONE 'UTC') AS utc_now_naive,
       MAX(created_at) AS max_created,
       (NOW() AT TIME ZONE 'UTC') - MAX(created_at) AS age
FROM signal_outcomes;

-- QS-03-A3: bias_composite_history adjudicator
SELECT (NOW() AT TIME ZONE 'UTC') AS utc_now_naive,
       MAX(created_at) AS max_created,
       (NOW() AT TIME ZONE 'UTC') - MAX(created_at) AS age
FROM bias_composite_history;

-- QS-03-A4: factor_readings adjudicator
SELECT (NOW() AT TIME ZONE 'UTC') AS utc_now_naive,
       MAX(created_at) AS max_created,
       (NOW() AT TIME ZONE 'UTC') - MAX(created_at) AS age
FROM factor_readings;

Reading key for EDGE (not for CC action): age of a live writer should be
small and positive. An age near minus-6h means that table IS write-shifted;
near +6h extra means a dead writer or a different defect. Raw values only.

-- QS-03-B1: orphaned outcome rows by status
SELECT o.outcome, COUNT(*) AS n
FROM signal_outcomes o
LEFT JOIN signals s ON s.signal_id = o.signal_id
WHERE s.signal_id IS NULL
GROUP BY 1 ORDER BY 2 DESC LIMIT 10;

-- QS-03-C1: diff-log transition matrix
SELECT old_outcome, new_outcome, COUNT(*) AS n
FROM signal_outcome_diff_log
GROUP BY 1, 2 ORDER BY 3 DESC LIMIT 30;

-- QS-03-C2: diff-log runs, dated
SELECT backfill_run_id, COUNT(*) AS n,
       MIN(created_at) AS first_row, MAX(created_at) AS last_row
FROM signal_outcome_diff_log
GROUP BY 1 ORDER BY 3 LIMIT 20;

-- QS-03-D1: strategy_health vocabulary + recency
SELECT source, window_days, COUNT(*) AS n, MAX(computed_at) AS latest
FROM strategy_health
GROUP BY 1, 2 ORDER BY 1, 2 LIMIT 60;

-- QS-03-E1: triton_flow_shadow extent + grading coverage
SELECT COUNT(*) AS n, MIN(fired_at) AS first_fire,
       MAX(fired_at) AS last_fire,
       COUNT(graded_at) AS n_graded, COUNT(fwd_ret_5d) AS n_fwd5
FROM triton_flow_shadow;

-- QS-03-F1: signal_forward_returns extent
SELECT COUNT(*) AS n, COUNT(DISTINCT signal_id) AS n_signals,
       MIN(computed_at) AS first_c, MAX(computed_at) AS last_c
FROM signal_forward_returns;

-- QS-03-G1: Phase-1 raw material — strategy x direction x outcome counts
SELECT s.strategy, s.direction, o.outcome, COUNT(*) AS n
FROM signals s
JOIN signal_outcomes o ON o.signal_id = s.signal_id
GROUP BY 1, 2, 3 ORDER BY 1, 2, 3 LIMIT 300;

END QS-03
